cmake_minimum_required(VERSION 3.24)
project(Birdie VERSION 1.0.0 LANGUAGES C CXX RC)
set(BIRDIE_VERSION_SUFFIX "")  # "" for a stable release; shown in the app and in the zip name

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Birdie builds with MSVC on Windows only (MSFS 2024 is Windows-only).")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# One self-contained exe: static CRT everywhere.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_compile_options(/W4 /permissive- /utf-8 "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)

# --- third party (header-only) ---------------------------------------------
add_library(nlohmann_json INTERFACE)
target_include_directories(nlohmann_json SYSTEM INTERFACE ${CMAKE_SOURCE_DIR}/third_party)

# miniz 3.0.2 (MIT): inflate for the Copernicus GeoTIFFs and reading the .zip files they come in (terrain/)
add_library(miniz STATIC third_party/miniz/miniz.c)
target_include_directories(miniz SYSTEM PUBLIC ${CMAKE_SOURCE_DIR}/third_party/miniz)
set_property(TARGET miniz PROPERTY COMPILE_OPTIONS /W0 /utf-8)  # replaces the project-wide /W4: not our code
target_compile_definitions(miniz PRIVATE _CRT_SECURE_NO_WARNINGS)

# WebView2 (Birdie's settings window): headers + the static loader, so Birdie.exe stays a single file. The runtime
# itself is part of Windows 11; without it the panel opens in the browser.
set(WEBVIEW2_DIR "${CMAKE_SOURCE_DIR}/third_party/webview2")
add_library(webview2 INTERFACE)
target_include_directories(webview2 SYSTEM INTERFACE "${WEBVIEW2_DIR}/include")
target_link_libraries(webview2 INTERFACE "${WEBVIEW2_DIR}/x64/WebView2LoaderStatic.lib" version)

# --- core: pure math + HUD scene, no SimConnect, no window ------------------
# Everything the parity tests check lives here, so they run without the sim.
add_library(birdie_core STATIC
  src/core/border.cpp
  src/core/camera.cpp
  src/core/geo.cpp
  src/core/model.cpp
  src/core/runway.cpp
  src/core/runway_heights.cpp
  src/core/settings.cpp
  src/core/version.cpp
  src/hud/scene.cpp
  src/hud/anim.cpp
  src/hud/terrain_draw.cpp
  src/terrain/terrain.cpp
  src/terrain/geotiff.cpp
  src/terrain/store.cpp
)
target_include_directories(birdie_core PUBLIC ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(birdie_core PUBLIC nlohmann_json miniz)
target_compile_definitions(birdie_core PUBLIC BIRDIE_VERSION="${PROJECT_VERSION}${BIRDIE_VERSION_SUFFIX}")

# --- render: Direct2D / DirectComposition overlay + off-screen rendering ----
add_library(birdie_render STATIC
  src/render/d2d.cpp
  src/terrain/terrain_gpu.cpp
)
target_link_libraries(birdie_render PUBLIC birdie_core)

# --- panel link (S5): HTTP/WebSocket pieces and the settings protocol, no sockets (unit-tested) ----------
add_library(birdie_panel STATIC
  src/app/ws.cpp
  src/app/panel_proto.cpp
  src/app/hotkeys.cpp
)
target_link_libraries(birdie_panel PUBLIC birdie_core)

# The whole in-sim panel package goes into Birdie.exe's resources (app/ingame.h): one exe serves the settings page
# on 127.0.0.1 and installs / updates / removes the package in the MSFS Community folder. Resource 2999 = the index
# (one relative path per line), 3000.. = the files in that order. Regenerated when files come or go
# (CONFIGURE_DEPENDS); a changed file rebuilds the resources (OBJECT_DEPENDS).
set(BIRDIE_PANEL_PKG "${CMAKE_SOURCE_DIR}/panel/birdie-panel")
file(GLOB_RECURSE BIRDIE_PANEL_FILES CONFIGURE_DEPENDS LIST_DIRECTORIES false RELATIVE "${BIRDIE_PANEL_PKG}"
     "${BIRDIE_PANEL_PKG}/*")
list(SORT BIRDIE_PANEL_FILES)
set(_pack_rc "// Generated by CMakeLists.txt: the in-sim panel package (app/ingame.h).\n")
set(_pack_index "")
set(_pack_deps "")
set(_pack_id 3000)
foreach(_f IN LISTS BIRDIE_PANEL_FILES)
  string(APPEND _pack_rc "${_pack_id} RCDATA \"${BIRDIE_PANEL_PKG}/${_f}\"\n")
  string(APPEND _pack_index "${_f}\n")
  list(APPEND _pack_deps "${BIRDIE_PANEL_PKG}/${_f}")
  math(EXPR _pack_id "${_pack_id} + 1")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/panel_pack/index.txt" CONTENT "${_pack_index}")
string(APPEND _pack_rc "2999 RCDATA \"${CMAKE_BINARY_DIR}/panel_pack/index.txt\"\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/panel_pack/panel_pack.rc" CONTENT "${_pack_rc}")
set_source_files_properties("${CMAKE_BINARY_DIR}/panel_pack/panel_pack.rc" PROPERTIES GENERATED TRUE
  OBJECT_DEPENDS "${_pack_deps};${CMAKE_BINARY_DIR}/panel_pack/index.txt")

# --- sim + app: Birdie.exe (needs the MSFS 2024 SDK for SimConnect) -----------
set(MSFS2024_SDK "C:/MSFS 2024 SDK" CACHE PATH "MSFS 2024 SDK root (SimConnect SDK inside)")
set(SIMCONNECT_DIR "${MSFS2024_SDK}/SimConnect SDK")
if(EXISTS "${SIMCONNECT_DIR}/include/SimConnect.h")
  add_library(birdie_sim STATIC
    src/sim/sim_client.cpp
  )
  target_include_directories(birdie_sim SYSTEM PUBLIC "${SIMCONNECT_DIR}/include")
  target_link_libraries(birdie_sim PUBLIC birdie_core "${SIMCONNECT_DIR}/lib/SimConnect.lib")

  add_executable(Birdie WIN32
    src/main.cpp
    src/app/app.cpp
    src/app/i18n.cpp
    src/app/ingame.cpp
    src/app/platform.cpp
    src/app/panel_server.cpp
    src/app/settings_window.cpp
    res/birdie.rc
    "${CMAKE_BINARY_DIR}/panel_pack/panel_pack.rc"
  )
  target_link_libraries(Birdie PRIVATE birdie_core birdie_render birdie_sim birdie_panel webview2 delayimp)
  # SimConnect.dll (import library, dynamic CRT) sits next to Birdie.exe; delay-loaded so that a missing DLL
  # gives a message instead of a loader error. The SDK's static SimConnect.lib needs /MD + OpenMP: not used.
  target_link_options(Birdie PRIVATE /DELAYLOAD:SimConnect.dll)
  add_custom_command(TARGET Birdie POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SIMCONNECT_DIR}/lib/SimConnect.dll" $<TARGET_FILE_DIR:Birdie>)
else()
  message(WARNING "SimConnect SDK not found at ${SIMCONNECT_DIR}: Birdie.exe is not built (set MSFS2024_SDK)")
endif()

# --- tests --------------------------------------------------------------------
option(BIRDIE_BUILD_TESTS "Build the parity tests" ON)
if(BIRDIE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/unit)
  add_subdirectory(tests/parity)
endif()
